# OnlineCoop additions to the game's script/modules/ModEventHandler.hps.
#
# The add-on overrides that script by name (the add-on mechanism of the
# official sample), but the game's script is not part of this project and is
# not redistributed. tools/InstallMod.ps1 and BunkerCoop.exe read it from the
# player's own installation and write the patched copy into mods/OnlineCoop
# at install time.
#
# One instruction per line: <op>|<anchor>|<line to insert>
#   after   insert after the line equal to <anchor>
#   body    insert as the first line of the block opened after <anchor>
#   before  insert before the line equal to <anchor>
# Anchors are compared after trimming whitespace and must match exactly one
# line. "\t" in the inserted text is a tab.
#
# sha256 is the game file these instructions were tested on. A different file
# (after a game update) is still patched if every anchor matches once.

body|void OnMapLeave(cLuxMap @apMap)|\t\tmCoop.OnMapLeave(apMap); // [COOP]
body|void Update(float afTimeStep)|\t\tmCoop.Update(afTimeStep); // [COOP]
