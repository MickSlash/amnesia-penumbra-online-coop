cmake_minimum_required(VERSION 3.20)
project(PenumbraOnlineCoop VERSION 0.3 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This verified build currently requires Windows and MSVC.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Configure with -A Win32: the original Newton runtime is 32-bit.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/build/dependencies")
include_directories(HPL1Engine/include OALWrapper/include "${DEPS}/include" "${DEPS}/include/AL" "${DEPS}/include/SDL")
link_directories("${DEPS}/lib/win32")
add_compile_definitions(WIN32 NOMINMAX _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS)
add_compile_options(/MP8 $<$<COMPILE_LANGUAGE:CXX>:/Zc:forScope->)
file(GLOB AS_SOURCES CONFIGURE_DEPENDS third_party/angelscript/source/*.cpp)
add_library(angelscript STATIC ${AS_SOURCES})
target_include_directories(angelscript PUBLIC third_party/angelscript/include)
target_compile_options(angelscript PRIVATE /Oy-)
add_subdirectory(third_party/enet EXCLUDE_FROM_ALL)
target_include_directories(enet PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/third_party/enet/include")
file(GLOB OAL_SOURCES CONFIGURE_DEPENDS OALWrapper/sources/*.cpp)
add_library(OALWrapper STATIC ${OAL_SOURCES})
target_link_libraries(OALWrapper PUBLIC libvorbisfile libvorbis libogg OpenAL32 SDL)
file(GLOB_RECURSE HPL_SOURCES CONFIGURE_DEPENDS HPL1Engine/sources/*.cpp HPL1Engine/sources/*.c)
list(FILTER HPL_SOURCES EXCLUDE REGEX "(HaptX\\.cpp|Fmod\\.cpp|posix\\.cpp)$")
add_library(HPL STATIC ${HPL_SOURCES})
target_link_libraries(HPL PUBLIC OALWrapper angelscript newton SDL SDL_image SDL_ttf libtheora libogg cg cgGL opengl32 glu32 winmm shell32 comdlg32 comctl32 ws2_32 legacy_stdio_definitions)
file(GLOB GAME_SOURCES CONFIGURE_DEPENDS game/*.cpp)
add_executable(PenumbraCoopGame WIN32 ${GAME_SOURCES})
target_sources(PenumbraCoopGame PRIVATE game/network/NetSession.cpp game/network/HostTransport.cpp)
target_link_libraries(PenumbraCoopGame PRIVATE HPL enet wintrust)
set_target_properties(PenumbraCoopGame PROPERTIES OUTPUT_NAME PenumbraCoop RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/build/bin")
enable_testing()
add_executable(NetworkTests tests/network/NetworkTests.cpp game/network/NetSession.cpp game/network/HostTransport.cpp)
target_link_libraries(NetworkTests PRIVATE enet ws2_32 winmm)
add_test(NAME ProtocolValidation COMMAND NetworkTests)
target_link_options(PenumbraCoopGame PRIVATE "/MAP:${CMAKE_CURRENT_SOURCE_DIR}/build/PenumbraCoop.map")
