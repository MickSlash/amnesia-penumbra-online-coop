cmake_minimum_required(VERSION 3.16)
project(AmnesiaNetworkTests LANGUAGES C CXX)
add_subdirectory(../../third_party/enet enet)
add_executable(NetworkTests NetworkTests.cpp ../../amnesia/src/game/network/NetSession.cpp)
target_include_directories(NetworkTests PRIVATE ../../third_party/enet/include)
target_compile_features(NetworkTests PRIVATE cxx_std_14)
target_link_libraries(NetworkTests PRIVATE enet)
if(WIN32)
  target_link_libraries(NetworkTests PRIVATE ws2_32 winmm)
endif()
enable_testing()
add_test(NAME ProtocolValidation COMMAND NetworkTests)
